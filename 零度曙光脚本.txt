Sub 主流程()

    Dim 字库0结果, 字库2结果, 字库3结果

    SetRowsNumber(0)

    字库0结果 = SetDictEx(0, "Attachment:字库.txt")

    TracePrint "【初始化】字库0加载结果: " & 字库0结果

    字库2结果 = SetDictEx(2, "Attachment:字库2.txt")

    TracePrint "【初始化】字库2加载结果: " & 字库2结果

    字库3结果 = SetDictEx(3, "Attachment:字库3.txt")

    TracePrint "【初始化】字库3加载结果: " & 字库3结果

    If 字库0结果 = 0 Or 字库2结果 = 0 Or 字库3结果 = 0 Then

        ShowMessage "字库加载失败，请检查字库文件", 5000, 0, 0

        EndScript

    End If

    Dim 包名, 连续闪退次数, 运行状态
    Dim intX, intY, 当前阶段, 提示信息, 需重新识别

    Dim 清理计数, 清理到东西

    包名 = "com.hi4fun.ldsgjlyq"

    连续闪退次数 = 0

    Dim 云手机兼容模式
    云手机兼容模式 = 1

    TracePrint "============================================"

    TracePrint "【主程序】启动完成，进入守护循环"

    TracePrint "============================================"

    Do

        If 云手机兼容模式 = 1 Then
            运行状态 = True
        Else
            运行状态 = Sys.IsRunning(包名)
        End If

        If 运行状态 = False Then
            连续闪退次数 = 连续闪退次数 + 1

            TracePrint "【闪退检测】游戏未运行，连续异常次数=" & 连续闪退次数

            ShowMessage "游戏闪退/未运行，正在重启 (" & 连续闪退次数 & ")", 2000, 0, 0

            Call 重启游戏(包名, 20000)

            If 连续闪退次数 >= 5 Then

                ShowMessage "连续闪退5次，暂停10秒", 3000, 0, 0

                Delay 10000

                连续闪退次数 = 0

            End If

        Else

            If 连续闪退次数 > 0 Then

                TracePrint "【进程检测】游戏已恢复运行，重置异常计数"

                连续闪退次数 = 0

            End If

            If 强制步骤1 = 1 Then

                TracePrint "【主流程】KillApp重启后强制走步骤1"

                强制步骤1 = 0

                当前阶段 = 1

            Else

                当前阶段 = 判断当前阶段()

            End If

            需重新识别 = 0

            If 当前阶段 = 2 Then

                提示信息 = "【步骤2】游戏主界面"

                TracePrint "【主流程】" & 提示信息

                ShowMessage 提示信息, 2000, 0, 0

                Call 步骤2_主线到20级()

                If 强制步骤1 = 1 Then

                    TracePrint "【主流程】步骤2因KillApp退出，跳过步骤3重新识别阶段"

                    步骤2重启后待补步骤3 = 1

                    需重新识别 = 1

                End If

            Else

                提示信息 = "【步骤1】登录/引导界面"

                TracePrint "【主流程】" & 提示信息

                ShowMessage 提示信息, 2000, 0, 0

                Call 步骤1_完全版()

                TracePrint "【主流程】步骤1结束，进入步骤2"

                Call 步骤2_主线到20级()

                If 强制步骤1 = 1 Then

                    TracePrint "【主流程】步骤2因KillApp退出，跳过步骤3重新识别阶段"

                    步骤2重启后待补步骤3 = 1

                    需重新识别 = 1

                End If

            End If

            If 需重新识别 = 0 Then

                If 步骤3已执行 = 0 Or 步骤2重启后待补步骤3 = 1 Then

                    TracePrint "【主流程】步骤2已真正结束（含确认B），开始执行步骤3"

                    Call 商店购买流程()

                    Call 步骤3_每日任务()

                    步骤3已执行 = 1

                    步骤2重启后待补步骤3 = 0

                    TracePrint "【主流程】步骤3执行完毕，清理窗口后执行获取水晶"

                    清理计数 = 0

                    Do

                        清理到东西 = 0

                        If 清理常见弹窗("清理窗口") = True Then
                            清理到东西 = 1
                        End If

                        FindStr 661, 28, 693, 60, "x", "FFFFFF-303030", 0.9, intX, intY

                        If intX > -1 And intY > -1 Then

                            TracePrint "【清理窗口】识别到【x】，点击"

                            Tap intX, intY

                            Delay 1000

                            清理到东西 = 1

                        End If

                        FindStr 133, 1200, 181, 1249, "X", "BFBFBF-303030", 0.8, intX, intY

                        If intX > -1 And intY > -1 Then

                            TracePrint "【清理窗口】识别到【对话X】，点击"

                            Tap intX, intY

                            Delay 500

                            清理到东西 = 1

                        End If

                        FindStr 487, 1042, 539, 1105, "首充", "CFCCCB-303030|FFFFFF-303030", 0.9, intX, intY

                        If intX > -1 And intY > -1 Then

                            TracePrint "【清理窗口】识别到【首充】，点击关闭"

                            Tap intX, intY

                            Delay 1000

                            清理到东西 = 1

                        End If

                        Call 检测并点击新手()

                        If 清理到东西 = 0 Then

                            清理计数 = 清理计数 + 1

                            If 清理计数 >= 2 Then

                                TracePrint "【清理窗口】连续2轮无弹窗，开始执行获取水晶"

                                Exit Do

                            End If

                        Else

                            清理计数 = 0

                        End If

                        Delay 500

                    Loop

                    If 脚本模式 = 2 Then

                        Call 六角色20级_循环()

                    Else

                        Call 获取水晶()

                    End If

                    TracePrint "【主流程】获取水晶执行完毕，后续循环仅做守护"

                Else

                    TracePrint "【主流程】步骤3已执行过，跳过"

                End If

            End If

        End If

        TracePrint "【主流程】循环结束，重新识别阶段"

    Loop

End Sub

Function 判断当前阶段()

    Dim intX, intY, ocr结果

    UseDict 2

    FindStr 519, 1128, 636, 1180, "完成", "6DFF94-303030|6AF696-303030|6CFB95-303030", 0.8, intX, intY

    If intX > -1 And intY > -1 Then

        TracePrint "【阶段识别】检测到【完成】 → 步骤2"

        判断当前阶段 = 2

        Exit Function

    End If

    FindStr 113, 598, 267, 872, "接取", "6AB3ED-303030|6CBAF7-303030", 0.9, intX, intY

    If intX > -1 And intY > -1 Then

        TracePrint "【阶段识别】检测到【接取】 → 步骤2"

        判断当前阶段 = 2

        Exit Function

    End If

    FindStr 528, 938, 634, 1227, "点击后可接取", "FFFFFF-303030|ECECEC-303030|D7D8D9-303030", 0.8, intX, intY

    If intX > -1 And intY > -1 Then

        TracePrint "【阶段识别】检测到【点击后可接取】 → 步骤2"

        判断当前阶段 = 2

        Exit Function

    End If

    Dim 距离OCR

    距离OCR = Ocr(517, 1168, 638, 1225, "DCF102-303030|BACC05-303030|B9CC05-303030", 0.8)

    If 距离OCR <> "" And IsNumeric(距离OCR) Then

        TracePrint "【阶段识别】检测到距离=" & 距离OCR & " → 步骤2"

        判断当前阶段 = 2

        Exit Function

    End If

    ocr结果 = Ocr(10, 513, 28, 556, "CFCCCB-303030|FFFFFF-303030", 0.8)

    If ocr结果 <> "" Then

        If IsNumeric(ocr结果) Then

            Dim 等级数字

            等级数字 = ocr结果 * 1

            If 等级数字 >= 1 And 等级数字 <= 100 Then

                TracePrint "【阶段识别】检测到等级=" & 等级数字 & " → 步骤2"

                判断当前阶段 = 2

                Exit Function

            End If

        End If

    End If

    FindStr 663, 52, 693, 170, "新手结束", "CFCCCB-303030|FFFFFF-303030", 0.9, intX, intY

    If intX > -1 And intY > -1 Then

        TracePrint "【阶段识别】检测到【新手结束】区域1 → 步骤2"

        判断当前阶段 = 2

        Exit Function

    End If

    FindStr 644, 983, 712, 1177, "新手", "FFFFFF-303030", 0.8, intX, intY

    If intX > -1 And intY > -1 Then

        TracePrint "【阶段识别】检测到【新手】区域2 → 步骤2"

        判断当前阶段 = 2

        Exit Function

    End If

    TracePrint "【阶段识别】未检测到步骤2特征 → 默认步骤1"

    判断当前阶段 = 1

End Function

Sub 重启游戏(包名, 等待时间)
   Dim i, 已退出, 已启动

   TracePrint "【重启】开始：先杀进程，等待退出后再重启"
   已退出 = 0
   已启动 = 0
   KillApp 包名
   Delay 2000

   For i = 1 To 8
       If Sys.IsRunning(包名) = False Then
           TracePrint "【重启】进程已退出"
           已退出 = 1
           Exit For
       End If
       TracePrint "【重启】等待进程退出，第 " & i & " 次"
       Delay 1000
   Next

   If 已退出 = 0 Then
       TracePrint "【重启】未确认进程退出，继续执行启动流程"
   End If

   TracePrint "【重启】重新启动游戏"
   RunApp 包名
   Delay 3000

   For i = 1 To 5
       If Sys.IsRunning(包名) = True Then
           TracePrint "【重启】已检测到进程重新启动"
           已启动 = 1
           Exit For
       End If
       TracePrint "【重启】等待进程启动，第 " & i & " 次"
       Delay 1000
   Next

   If 已启动 = 0 Then
       TracePrint "【重启】未确认进程启动，云手机可能误判，继续等待进入游戏"
   End If

   Delay 等待时间

   强制步骤1 = 1
   重启后步骤1 = 1
   TracePrint "【重启】重启完成，返回主流程后强制走步骤1"
End Sub

Function 启动应用(包名, 等待时间)
   Dim 运行状态, i, 已启动

   运行状态 = Sys.isRunning(包名)
   If 运行状态 = True Then
       TracePrint "【启动】应用已在运行，无需等待"
       启动应用 = True
       Exit Function
   End If

   已启动 = 0
   For i = 1 To 2
       TracePrint "【启动】未确认进程运行，第 " & i & " 次尝试拉起应用"
       RunApp 包名
       Delay 3000

       运行状态 = Sys.isRunning(包名)
       If 运行状态 = True Then
           TracePrint "【启动】已检测到进程运行"
           已启动 = 1
           Exit For
       End If
   Next

   If 已启动 = 0 Then
       TracePrint "【启动】未确认进程运行，云手机可能误判，继续等待进入游戏"
   End If

   Delay 等待时间
   启动应用 = True
End Function

Function 输入英文或数字()

   Dim 字符, 结果, 重试次数

   Randomize

   字符 = "abcdefghijklmnopqrstuvwxyz0123456789"

   重试次数 = 0
   Do
       结果 = Mid(字符, Int(Rnd() * 36) + 1, 1)
       重试次数 = 重试次数 + 1
       If 结果 <> 上次输入字符 Then Exit Do
       If 重试次数 >= 5 Then Exit Do
   Loop

   TracePrint "【输入】随机字符: " & 结果

   InputText 结果

   上次输入字符 = 结果

   Delay 500

End Function

' 获取当前默认输入法
Function getNowIme()
    getNowIme = zm.Execute("settings get secure default_input_method")
End Function

' 设置按键输入法为默认，并卸载所有其他输入法（root）
Function setImeEx(packageName)
    Dim packageNameIme, allIme, imeArr, k, imeId, pkgName, slashPos, ret, curIme
    If packageName = "" Then
        packageName = 按键输入法包名
    End If
    packageNameIme = zm.Execute("ime list -s -a |grep " & packageName & "|head -n 1")
    If packageNameIme = "" Then
        ShowMessage "获取按键输入法失败"
        TracePrint "【换账号】获取按键输入法失败，包名：" & packageName
        Exit Function
    End If
    allIme = zm.Execute("ime list -s -a")
    已卸载输入法列表 = ""
    imeArr = Split(allIme, Chr(10))
    For k = 0 To UBound(imeArr)
        imeId = Trim(imeArr(k))
        If imeId <> "" And InStr(1, imeId, packageName) = 0 Then
            slashPos = InStr(1, imeId, "/")
            If slashPos > 0 Then
                pkgName = Left(imeId, slashPos - 1)
            Else
                pkgName = imeId
            End If
            ret = zm.Execute("su -c ""pm uninstall --user 0 " & pkgName & """")
            zm.Execute("su -c ""am force-stop " & pkgName & """")
            TracePrint "【换账号】已卸载输入法：" & pkgName
            If 已卸载输入法列表 = "" Then
                已卸载输入法列表 = pkgName
            Else
                已卸载输入法列表 = 已卸载输入法列表 & "|" & pkgName
            End If
        End If
    Next
    zm.Execute("su -c ""settings put secure default_input_method " & packageNameIme & """")
    zm.Execute("su -c ""content insert --uri content://settings/secure --bind name:s:default_input_method --bind value:s:" & packageNameIme & """")
    Delay 800
    curIme = zm.Execute("settings get secure default_input_method")
    If InStr(1, curIme, packageName) > 0 Then
        TracePrint "【换账号】按键输入法启用成功：" & curIme
    Else
        TracePrint "【换账号】警告：默认输入法切换失败，当前=" & curIme
    End If
End Function

' 恢复原始输入法并重新安装已卸载的输入法
Function restoreAllIme(typewriting)
    Dim uninstalledArr, k, pkgName, ret, curIme
    zm.Execute("su -c ""settings put secure default_input_method " & typewriting & """")
    zm.Execute("su -c ""content insert --uri content://settings/secure --bind name:s:default_input_method --bind value:s:" & typewriting & """")
    If 已卸载输入法列表 <> "" Then
        uninstalledArr = Split(已卸载输入法列表, "|")
        For k = 0 To UBound(uninstalledArr)
            pkgName = Trim(uninstalledArr(k))
            If pkgName <> "" Then
                ret = zm.Execute("su -c ""pm install-existing " & pkgName & """")
                TracePrint "【换账号】已恢复输入法：" & pkgName
            End If
        Next
    End If
    Delay 500
    curIme = zm.Execute("settings get secure default_input_method")
    TracePrint "【换账号】输入法已恢复：" & curIme
End Function

' 按键模拟输入（绕过输入法，不能输入中文）
Function KeyPress_Input(s)
    Dim i, ch
    If s = "" Then
        TracePrint "【换账号】KeyPress_Input：输入内容为空"
        Exit Function
    End If
    For i = 1 To Len(s)
        ch = Mid(s, i, 1)
        If ch = " " Then
            KeyPress "Space"
        Else
            KeyPress ch
        End If
        Delay 20
    Next
End Function
